Problem 1: Three ways to sum to n

Method A: Iterative Approach (For Loop)

var sum_to_n_a = function(n) { let sum = 0; for (let i = 1; i <= n; i++) { sum += i; } return sum; };
Time Complexity: O(n) | Space Complexity: O(1)

Method B: Mathematical Formula (Most Efficient)

var sum_to_n_b = function(n) { return n * (n + 1) / 2; };
Time Complexity: O(1) | Space Complexity: O(1)

Method C: Recursive Approach

var sum_to_n_c = function(n) { if (n <= 0) return 0; if (n === 1) return 1; return n + sum_to_n_c(n - 1); };
Time Complexity: O(n) | Space Complexity: O(n)
